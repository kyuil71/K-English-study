휴대폰에서 로그인이 자주 끊기면 이 파일을 Firebase Hosting(*.web.app)에 올려 주세요. 같은 도메인에서는 팝업 대신 리디렉션 방식을 써서 훨씬 안정적입니다. 이 앱은 Cloud Storage를 쓰지 않고 Firestore만 사용하므로 무료 요금제에서도 동작합니다. 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null
                         && request.auth.uid == userId;
    }
  }
}
